using System.Diagnostics;
using System.Reflection;
using System.Text.Json;
using LiteMonitor;
using LiteMonitor.src.Core;
using LiteMonitor.src.Plugins;
using LiteMonitor.Rendering.Configuration;
using LiteMonitor.ThemeEditor;

static class ReviewProbe
{
    const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    static T Field<T>(object o, string n) => (T)o.GetType().GetField(n, Private)!.GetValue(o)!;
    static void Pump(int ms)
    {
        var w = Stopwatch.StartNew();
        while (w.ElapsedMilliseconds < ms) { Application.DoEvents(); Thread.Sleep(5); }
    }
    [STAThread]
    static void Main(string[] args)
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        Application.EnableVisualStyles();
        string output = Path.GetFullPath(args[0]);
        Directory.CreateDirectory(output);
        string td = Path.Combine(output, "themes"); Directory.CreateDirectory(td);
        string sd = Path.Combine(output, "settings"); Directory.CreateDirectory(sd);
        SettingsHelper.PathOverride = sd;
        ThemeDocumentStore.SetThemeDirForTests(td);
        ThemeDocumentStore.SetRecoveryDirForTests(Path.Combine(output, "recovery"));
        var cfg = new Settings { GpuThemeEffectsEnabled = false, TaskbarCustomStyle = true,
            TaskbarColorLabel = "#123456", TaskbarSingleLine = true };
        cfg.TrySave(); Settings.Load(true);
        var theme = new Theme { Name = "ReviewGpu", Color = new ColorConfig { Background = "#112233" },
            Rendering = new ThemeRenderingConfig { Backend = "gpu", Material = "cyber_punk",
                FallbackStyle = new ThemeFallbackStyle { Background = "#AABBCC" } } };
        theme.BuildFonts();
        File.WriteAllText(Path.Combine(td, "ReviewGpu.json"), JsonSerializer.Serialize(theme, ThemeManager.CreateSerializerOptions()));
        var results = new Dictionary<string, object?>();
        if (args.Length > 1 && args[1] == "dpi")
        {
            cfg.GpuThemeEffectsEnabled = true;
            typeof(Settings).GetField("_instance", BindingFlags.Static | BindingFlags.NonPublic)!.SetValue(null, cfg);
            using var frame = new Form { ClientSize = new Size(500, 700) };
            using var preview = new ThemePreviewControl { Dock = DockStyle.Fill };
            frame.Controls.Add(preview); frame.Show();
            preview.SetTheme(theme, ThemeEditTarget.GpuMain); Pump(120);
            var gpu = Field<GpuThemePreviewControl>(preview, "_gpuPreview");
            var first = Field<Theme>(gpu, "_theme");
            var firstState = new { dpi = preview.DeviceDpi, gpu.IsActive, gpu.Error,
                layoutScale = first.Layout.LayoutScale, rowHeight = first.Layout.RowHeight };
            preview.SetTheme(theme, ThemeEditTarget.GpuMain);
            var secondRenderer = Field<object>(gpu, "_renderer");
            var immediate = new { gpu.IsActive, gpu.Error, preview.CurrentGeneration,
                rendererGenerations = Field<object>(secondRenderer, "_generations").ToString() };
            Pump(120);
            var next = Field<Theme>(gpu, "_theme");
            var report = JsonSerializer.Serialize(new { first = firstState, immediatelyAfterRefresh = immediate,
                afterUnchangedRefresh = new { dpi = preview.DeviceDpi, gpu.IsActive, gpu.Error,
                    layoutScale = next.Layout.LayoutScale, rowHeight = next.Layout.RowHeight },
                preview.DeviceCreationCount, preview.SessionUpdateCount }, new JsonSerializerOptions { WriteIndented = true });
            Console.WriteLine(report); File.WriteAllText(Path.Combine(output, "dpi-observations.json"), report);
            frame.Close(); theme.DisposeFonts(); return;
        }
        using (var form = new ThemeStudioForm(appSettings: cfg))
        {
            form.Show(); form.OpenThemeForTesting("ReviewGpu"); Pump(60);
            var preview = form.PreviewControlForTesting!;
            results["automatic_fallback"] = new { target = preview.RenderedTarget.ToString(),
                gpu = preview.IsGpuPreviewActive, background = preview.RenderedTheme!.Color.Background,
                expected = "#AABBCC", status = form.RouteStatusTextForTesting };
            form.SelectPreviewShapeForTesting(1); Pump(60);
            using (var bitmap = new Bitmap(preview.Width, preview.Height))
            { preview.DrawToBitmap(bitmap, preview.ClientRectangle); bitmap.Save(Path.Combine(output, "automatic-fallback-horizontal.png")); }
            results["horizontal_automatic_fallback"] = new { status = form.RouteStatusTextForTesting,
                horizontalControlExists = form.HorizontalPreviewForTesting != null,
                painterLayout = Field<object>(preview, "_layout").GetType().Name,
                background = preview.RenderedTheme!.Color.Background };
            form.SelectPreviewShapeForTesting(2); Pump(30);
            var tp = form.TaskbarPreviewForTesting!;
            var pcfg = Field<Settings>(tp, "_settings");
            results["taskbar_settings"] = new { expectedCustom = cfg.TaskbarCustomStyle,
                actualCustom = pcfg.TaskbarCustomStyle, expectedLabel = cfg.TaskbarColorLabel,
                actualLabel = pcfg.TaskbarColorLabel, expectedSingle = cfg.TaskbarSingleLine,
                actualSingle = pcfg.TaskbarSingleLine };
            form.SelectPreviewShapeForTesting(0);
            int activeCount = 0;
            for (int i = 1; i <= 200; ++i)
            { form.TargetComboForTesting.SelectedIndex = i % 2 == 0 ? 1 : 0; Application.DoEvents();
                if (preview.IsGpuPreviewActive) activeCount++; }
            results["resource_switch_fixture_with_gpu_disabled"] = new { iterations = 200, activeCount,
                deviceCreations = preview.DeviceCreationCount, effectsEnabled = Settings.Load().GpuThemeEffectsEnabled };
            form.Close();
        }
        using (var host = new Form { ClientSize = new Size(400, 500) })
        using (var gpu = new GpuThemePreviewControl(theme, ThemePreviewControl.CreateMockGroups(),
            new GpuStartupGuard(Path.Combine(output, "guard"), AppContext.BaseDirectory)) { Size = new Size(300, 420) })
        {
            host.Controls.Add(gpu); host.Show(); Pump(250);
            int before = gpu.PresentedFrameCount;
            host.WindowState = FormWindowState.Minimized; Pump(500);
            results["minimized_gpu"] = new { gpu.IsActive, host.Visible, state = host.WindowState.ToString(),
                childVisible = gpu.Visible, before, after = gpu.PresentedFrameCount,
                framesWhileMinimized = gpu.PresentedFrameCount - before };
            host.WindowState = FormWindowState.Normal; Pump(60);
            typeof(Control).GetMethod("RecreateHandle", Private)!.Invoke(gpu, null); Pump(100);
            results["gpu_recreate"] = new { gpu.IsActive, gpu.Error, gpu.DeviceCreationCount };
            host.Close();
        }
        var template = JsonSerializer.Deserialize<PluginTemplate>(File.ReadAllText("resources/plugins/vps_traffic.json"))!;
        var hosts = PluginProcessor.GetDeclaredHosts(template);
        var headers = new Dictionary<string,string> { ["X-API-Token"] = "review-dummy-token" };
        bool permitted = true;
        try { PluginSandbox.EnsureCredentialTargetAllowed("https://unlisted.example/api", headers, hosts); }
        catch { permitted = false; }
        results["vps_credential_policy"] = new { declaredHosts = hosts.ToArray(),
            headerRecognizedAsCredential = PluginSandbox.HasCredentialHeader(headers),
            dummySecretPermittedToUndeclaredHost = permitted,
            tokenInputType = template.Inputs!.Single(i => i.Key == "token").Type };
        var json = JsonSerializer.Serialize(results, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(Path.Combine(output, "observations.json"), json); Console.WriteLine(json);
        theme.DisposeFonts();
    }
}
