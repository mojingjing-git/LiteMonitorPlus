    [STAThread]
    static void Main(string[] args)
    {
        if (args.Contains("gpu")) { Gpu(); return; }
        if (args.Contains("invalid-ui")) { InvalidUi(); return; }
        if (args.Contains("web-result")) { WebResult(); return; }
        if (args.Contains("closure")) { Closure(); return; }
        if (args.Contains("diagnostic-ui")) { DiagnosticUi(); return; }
        if (args.Contains("fallback-ui")) { FallbackUi(); return; }
        if (args.Contains("apply-fault")) { ApplyFault(); return; }
        string dir = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "../../../../observations"));
        Directory.CreateDirectory(dir);
        var facts = new Dictionary<string, object?>();
        var now = new DateTimeOffset(2026, 10, 27, 4, 0, 0, TimeSpan.Zero);
        string themePath = Path.Combine(dir, "bom-theme.json");
        File.WriteAllText(themePath, "{\"Name\":\"BOM test\"}", new UTF8Encoding(true));
        var bom = ThemeManager.Inspect(themePath, "bom");
        facts["bomTheme"] = new { oldReaderParses = JsonSerializer.Deserialize<Theme>(File.ReadAllText(themePath)) is not null, bom.Success, bom.Error };
        bom.Theme?.DisposeFonts();

        var buffer = new ShortTermTrendBuffer();
        var sample = new MetricSample("CPU.Temp", "cpu-A", 95, "C", SampleQuality.Good, null, 0);
        buffer.Ingest(new HardwareSnapshot(1, 1, 1, new[] { sample }, TimeSpan.Zero, 1000), now);
        buffer.Ingest(new HardwareSnapshot(2, 1, 1, new[] { sample with { SampleAgeMs = 1000 } }, TimeSpan.Zero, 2000), now.AddSeconds(1));
        buffer.Ingest(new HardwareSnapshot(3, 1, 1, new[] { sample with { Value = null, Quality = SampleQuality.Missing } }, TimeSpan.Zero, 3000), now.AddSeconds(2));
        facts["shortTrend"] = new { points = buffer.GetSeries("CPU.Temp", "cpu-A"), stats = buffer.GetStats() };

        var runtime = new AlertRuntime();
        var rule = new AlertRule("r", "CPU", "CPU.Temp", AlertDirection.Above, 90, 80,
            TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10), TimeSpan.FromSeconds(30), TimeSpan.FromMinutes(3));
        var rules = new[] { rule };
        runtime.Evaluate(rules, _ => 95, now);
        facts["cachedAlert"] = new { transitions = runtime.Evaluate(rules, _ => 95, now.AddSeconds(6)), state = runtime.GetState("r").ToString(), runtime.CurrentBatchId };
        runtime.Evaluate(new[] { rule with { } }, _ => 95, now.AddSeconds(7));
        var copiedRules = new[] { rule with { } };
        runtime.Evaluate(copiedRules, _ => 95, now.AddSeconds(8));
        facts["alertResetByIdenticalRules"] = runtime.Evaluate(copiedRules, _ => 95, now.AddSeconds(14));
        var evaluator = new AlertEvaluator();
        AlertSample AlertAt(int sec, long batch, double? value) => new("CPU.Temp", "cpu-A", value, now.AddSeconds(sec), "session", batch);
        evaluator.Evaluate(rule, AlertAt(0, 1, 95));
        evaluator.Evaluate(rule, AlertAt(6, 2, 95));
        evaluator.Evaluate(rule, AlertAt(7, 3, 70));
        var recoveringGap = evaluator.Evaluate(rule, AlertAt(100, 4, 70));
        facts["alertRecoveringGap"] = new { transitions = recoveringGap, state = evaluator.GetState("r", "cpu-A").ToString(), incident = evaluator.GetIncidentId("r", "cpu-A") };

        string sp = Path.Combine(dir, "profiles.json");
        var store = new ScenarioProfileStore(sp);
        store.Save(ScenarioProfile.Create("A", new Settings()), 0);
        store.Save(ScenarioProfile.Create("B", new Settings()), 0);
        File.WriteAllText(sp, "{broken");
        var first = store.LoadAll();
        var second = store.LoadAll();
        facts["scenarioBackup"] = new { first = first.Select(p => p.Name), second = second.Select(p => p.Name), primaryExists = File.Exists(sp) };

        string fp = Path.Combine(dir, "future-profiles.json");
        var future = ScenarioProfile.Create("future", new Settings());
        future.SchemaVersion = 99;
        File.WriteAllText(fp, JsonSerializer.Serialize(new[] { future }, ScenarioProfileJson.Options));
        var fs = new ScenarioProfileStore(fp);
        var futureLoad = fs.LoadAll();
        var futureSave = fs.Save(ScenarioProfile.Create("replacement", new Settings()), 0);
        facts["futureScenario"] = new { loaded = futureLoad.Count, saveAccepted = futureSave.Success, activeNames = fs.LoadAll().Select(p => p.Name) };

        var live = new Settings { HideTrayIcon = true, HideMainForm = false, ShowTaskbar = true, ClickThrough = false, TaskbarClickThrough = false };
        var lockedProfile = ScenarioProfile.Create("click-through", new Settings { HideMainForm = false, ShowTaskbar = true, ClickThrough = true, TaskbarClickThrough = true });
        var preflight = ScenarioPreflight.Inspect(lockedProfile, live, _ => true);
        var lockedCandidate = live.DeepClone();
        ScenarioProjection.ApplyTo(lockedCandidate, lockedProfile.Settings);
        facts["scenarioInteractionLockout"] = new { preflight.CanApply, lockedCandidate.HideTrayIcon, lockedCandidate.HideMainForm, lockedCandidate.ShowTaskbar, lockedCandidate.ClickThrough, lockedCandidate.TaskbarClickThrough, noInteractiveEntry = lockedCandidate.HideTrayIcon && (lockedCandidate.HideMainForm || lockedCandidate.ClickThrough) && (!lockedCandidate.ShowTaskbar || lockedCandidate.TaskbarClickThrough) };

        var gpuSettings = new Settings { MonitoredGpuCount = 1 };
        AlertRuleMigration.Apply(gpuSettings, 1);
        gpuSettings.MonitoredGpuCount = 2;
        bool remigrated = AlertRuleMigration.Apply(gpuSettings, 2);
        facts["gpuAlertMigration"] = new { remigrated, gpuSettings.MonitoredGpuCount, gpuRules = gpuSettings.AlertRules.Where(r => r.MetricKey.StartsWith("GPU.")).Select(r => r.MetricKey) };

        var storeEvents = new DiagnosticEventStore();
        foreach (string id in new[] { "cpu", "gpu" })
            storeEvents.Add(new DiagnosticEvent(now, "alerts", DiagnosticSeverity.Warning, "alert_triggered", null, "s", new Dictionary<string, string> { ["ruleId"] = id, ["deviceId"] = id, ["state"] = "Active", ["reason"] = "trigger_duration_met" }));
        facts["alertDiagnostic"] = storeEvents.Snapshot();

        var snapshot = new DiagnosticSnapshot(now, new[] { new DiagnosticModuleSnapshot("apply", DiagnosticModuleState.Error, "failed", "failed", now, new Dictionary<string, string> { ["module.ThemeRuntime"] = @"failed: Could not open G:\Users\Alice\private-theme.json (host 192.0.2.1)" }) });
        string zipPath = Path.Combine(dir, "privacy.zip");
        var exported = new DiagnosticPackageService().Export(zipPath, new DiagnosticPackageInput(snapshot, Array.Empty<DiagnosticEvent>(), "test"));
        using (var zip = ZipFile.OpenRead(zipPath))
        using (var reader = new StreamReader(zip.GetEntry("diagnostic.json")!.Open()))
            facts["privacy"] = new { exported.Success, json = reader.ReadToEnd() };

        var hist = new HardwareHistoryData { Points = new() { new HardwareHistoryPoint { Time = now.LocalDateTime, Values = new() { ["GPU.0.Temp"] = new HardwareHistoryValue { Avg = 65, Min = 30, Max = 99, Count = 6 } } } } };
        string old = Path.Combine(dir, "legacy-history.json");
        File.WriteAllText(old, JsonSerializer.Serialize(hist));
        var hs = new TrendHistoryV2Store(Path.Combine(dir, "history"), clock: () => now);
        hs.ImportV1(old);
        facts["historyImport"] = hs.ReadSince(now.AddMinutes(-1));
        facts["powerCaps"] = new { ecoAt5 = PowerPolicy.ResolveDisplay(PowerMode.Eco, false, true, 5), autoAt5 = PowerPolicy.ResolveDisplay(PowerMode.Auto, true, true, 5), ecoAnimation = PowerPolicy.ResolveAnimationFps(PowerMode.Eco, false, 60), autoAnimation = PowerPolicy.ResolveAnimationFps(PowerMode.Auto, true, 60) };
        string json = JsonSerializer.Serialize(facts, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(Path.Combine(dir, "observations.json"), json);
        Console.WriteLine(json);
    }

    static void Pump(int milliseconds)
    {
        var sw = Stopwatch.StartNew();
        while (sw.ElapsedMilliseconds < milliseconds) { Application.DoEvents(); Thread.Sleep(5); }
    }

    static void Gpu()
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        string gpuThemePath = Directory.GetFiles(Path.Combine(Environment.CurrentDirectory, "resources", "themes"), "*_GPU.json").First();
        Theme theme = ThemeManager.Inspect(gpuThemePath).Theme ?? throw new InvalidDataException(gpuThemePath);
        string scratch = Path.Combine(Path.GetTempPath(), "lm-review-gpu-" + Guid.NewGuid().ToString("N"));
        var guard = new GpuStartupGuard(scratch, AppContext.BaseDirectory);
        using var form = new Form { ClientSize = new Size(500, 500) };
        using var preview = new GpuThemePreviewControl(ThemePreviewControl.CreatePreviewTheme(theme, 1.75f), ThemePreviewControl.CreateMockGroups(), guard);
        preview.Dock = DockStyle.Fill;
        form.Controls.Add(preview);
        form.Show();
        Pump(300);
        var observations = new List<object>();
        for (int i = 0; i < 3; i++)
        {
            bool updated = preview.UpdateTheme(ThemePreviewControl.CreatePreviewTheme(theme, 1.75f));
            Pump(180);
            observations.Add(new { i, updated, preview.IsActive, preview.Error, preview.PresentedFrameCount, preview.DeviceCreationCount, preview.SessionUpdateCount, preview.CurrentGeneration, preview.DeviceDpi });
        }
        int before = preview.PresentedFrameCount;
        form.WindowState = FormWindowState.Minimized;
        Pump(220);
        int minimized = preview.PresentedFrameCount;
        form.WindowState = FormWindowState.Normal;
        Pump(220);
        Console.WriteLine(JsonSerializer.Serialize(new { observations, before, minimized, restored = preview.PresentedFrameCount, restoredActive = preview.IsActive, restoredError = preview.Error, deviceCreations = preview.DeviceCreationCount, finalGeneration = preview.CurrentGeneration }, new JsonSerializerOptions { WriteIndented = true }));
        form.Close();
        theme.DisposeFonts();
    }

    static void InvalidUi()
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        Settings.Load().GpuThemeEffectsEnabled = true;
        string root = Path.Combine(Path.GetTempPath(), "lm-review-invalid-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(root);
        File.WriteAllText(Path.Combine(root, "invalid.json"), "{\"name\":\"invalid\",\"rendering\":{\"backend\":\"gpu\",\"material\":\"invalid\"}}");
        const BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic;
        typeof(ThemeDocumentStore).GetMethod("SetThemeDirForTests", BindingFlags.Static | BindingFlags.NonPublic)!.Invoke(null, new object[] { root });
        using var form = new ThemeStudioForm();
        form.Show();
        typeof(ThemeStudioForm).GetMethod("OpenThemeForTesting", flags)!.Invoke(form, new object[] { "invalid" });
        var preview = (ThemePreviewControl)typeof(ThemeStudioForm).GetField("_preview", flags)!.GetValue(form)!;
        int count = 0;
        preview.GpuPreviewUnavailable += (_, _) => { if (++count >= 5) form.Close(); };
        typeof(ThemeStudioForm).GetMethod("RefreshPreviewForTesting", flags)!.Invoke(form, null);
        var sw = Stopwatch.StartNew();
        while (!form.IsDisposed && sw.ElapsedMilliseconds < 1000) { Application.DoEvents(); Thread.Sleep(1); }
        Console.WriteLine(JsonSerializer.Serialize(new { failureCallbacksWithoutUserInput = count, elapsedMs = sw.ElapsedMilliseconds, form.IsDisposed }));
        if (!form.IsDisposed) form.Close();
    }

    static void Closure()
    {
        var facts = new Dictionary<string, object?>();
        var now = new DateTimeOffset(2026, 10, 27, 4, 0, 0, TimeSpan.Zero);
        var rule = new AlertRule("cpu", "CPU", "CPU.Temp", AlertDirection.Above, 80, 70,
            TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10), TimeSpan.FromSeconds(30), TimeSpan.FromMinutes(3));
        HardwareSnapshot Snapshot(long batch, double mono, float? value, double age = 0, string? id = "cpu-A", SampleQuality quality = SampleQuality.Good) =>
            new(batch, 1, 1, new[] { new MetricSample("CPU.Temp", id, value, "C", quality, null, age) }, TimeSpan.Zero, mono, 1);

        var cached = new AlertRuntime();
        var rules = new[] { rule };
        cached.Evaluate(rules, Snapshot(1, 1000, 90), now);
        facts["sameSourceDifferentBatch"] = cached.Evaluate(rules, Snapshot(2, 2000, 90, age: 1000), now.AddSeconds(6));

        var outage = new AlertRuntime();
        outage.Evaluate(rules, Snapshot(1, 1000, 90), now);
        var missing = outage.Evaluate(rules, Snapshot(2, 2000, null, id: null, quality: SampleQuality.Missing), now.AddSeconds(1));
        var resumed = outage.Evaluate(rules, Snapshot(3, 7000, 90), now.AddSeconds(6));
        facts["missingIdentity"] = new { missing, resumed, originalState = outage.GetState("cpu", "cpu-A").ToString() };

        var changed = new AlertRuntime();
        changed.Evaluate(rules, Snapshot(1, 1000, 90), now);
        var higherRule = rule with { TriggerThreshold = 100, RecoveryThreshold = 90 };
        facts["ruleDefinitionChanged"] = changed.Evaluate(new[] { higherRule }, Snapshot(2, 7000, 110), now.AddSeconds(6));

        facts["redaction"] = new Dictionary<string, bool>
        {
            ["ipv6Allowed"] = DiagnosticRedaction.IsSafeValue("2001:db8::1234"),
            ["colonWindowsPathAllowed"] = DiagnosticRedaction.IsSafeValue(@"failed:G:\Users\Alice\private.json"),
            ["otherUnixRootAllowed"] = DiagnosticRedaction.IsSafeValue("/srv/private/alice/data.json"),
        };
        string dir = Path.Combine(Path.GetTempPath(), "lm-closure-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(dir);
        var snapshot = new DiagnosticSnapshot(now, new[] { new DiagnosticModuleSnapshot("apply", DiagnosticModuleState.Error, "failed", "failed", now,
            new Dictionary<string, string> { ["module.v6"] = "2001:db8::1234", ["module.path"] = @"failed:G:\Users\Alice\private.json" }) });
        string zipPath = Path.Combine(dir, "diagnostics.zip");
        var export = new DiagnosticPackageService().Export(zipPath, new DiagnosticPackageInput(snapshot, Array.Empty<DiagnosticEvent>(), "test"));
        using (var zip = ZipFile.OpenRead(zipPath))
        using (var reader = new StreamReader(zip.GetEntry("diagnostic.json")!.Open()))
            facts["export"] = new { export.Success, json = reader.ReadToEnd() };

        var encodings = new Dictionary<string, object>();
        foreach (var encoding in new Encoding[] { new UTF8Encoding(true), Encoding.Unicode })
        {
            string themePath = Path.Combine(dir, encoding.CodePage + ".json");
            File.WriteAllText(themePath, "{\"name\":\"BOM accepted\"}", encoding);
            var inspected = ThemeManager.Inspect(themePath);
            encodings[encoding.WebName] = new { inspected.Success, inspected.Error, hashMatches = inspected.SourceHash == Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(File.ReadAllBytes(themePath))) };
            inspected.Theme?.DisposeFonts();
        }
        facts["fixedBom"] = encodings;
        var history = new HardwareHistoryData { Points = new() { new HardwareHistoryPoint { Time = now.LocalDateTime, Values = new() { ["GPU.0.Temp"] = new HardwareHistoryValue { Avg = 65, Max = 99, Min = 30, Count = 6 } } } } };
        string historyPath = Path.Combine(dir, "history.json");
        File.WriteAllText(historyPath, JsonSerializer.Serialize(history));
        var historyStore = new TrendHistoryV2Store(Path.Combine(dir, "v2"), clock: () => now);
        historyStore.ImportV1(historyPath);
        facts["fixedHistoryStatistics"] = historyStore.ReadSince(now.AddMinutes(-1));

        var scenarioStore = new ScenarioProfileStore(Path.Combine(dir, "profiles.json"));
        scenarioStore.Save(ScenarioProfile.Create("A", new Settings()), 0);
        scenarioStore.Save(ScenarioProfile.Create("B", new Settings()), 0);
        File.WriteAllText(scenarioStore.FilePath, "{broken");
        int firstRead = scenarioStore.LoadAll().Count;
        facts["fixedScenarioBackup"] = new { firstRead, secondRead = scenarioStore.LoadAll().Count };
        var currentSettings = new Settings { HideTrayIcon = true, ShowTaskbar = true, HideMainForm = false };
        var lockedProfile = ScenarioProfile.Create("locked", new Settings { ClickThrough = true, TaskbarClickThrough = true, ShowTaskbar = true });
        facts["fixedScenarioSafety"] = ScenarioPreflight.Inspect(lockedProfile, currentSettings, _ => true);

        var future = ScenarioProfile.Create("future", new Settings());
        future.SchemaVersion = 99;
        var futureStore = new ScenarioProfileStore(Path.Combine(dir, "future.json"));
        File.WriteAllText(futureStore.FilePath, JsonSerializer.Serialize(new[] { future }));
        futureStore.LoadAll();
        facts["fixedFutureSchema"] = new { futureStore.IsReadOnly, primaryPreserved = File.Exists(futureStore.FilePath), save = futureStore.Save(ScenarioProfile.Create("new", new Settings()), 0).Success };
        var diagnosticStore = new DiagnosticEventStore();
        foreach (string id in new[] { "cpu", "gpu" })
            diagnosticStore.Add(new DiagnosticEvent(now, "alerts", DiagnosticSeverity.Warning, "alert_triggered", null, "session", new Dictionary<string, string> { ["ruleId"] = id, ["deviceId"] = id }));
        facts["fixedDiagnosticIdentity"] = diagnosticStore.Snapshot();
        facts["fixedPowerCaps"] = new { eco = PowerPolicy.ResolveDisplay(PowerMode.Eco, false, true, 5).Fps, auto = PowerPolicy.ResolveDisplay(PowerMode.Auto, true, true, 5).Fps, animation = PowerPolicy.ResolveAnimationFps(PowerMode.Eco, true, 5) };

    static void DiagnosticUi()
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        int reads = 0;
        DiagnosticHub.Current.SnapshotProvider = () => { reads++; return new DiagnosticSnapshot(DateTimeOffset.UtcNow,
            new[] { new DiagnosticModuleSnapshot("test", DiagnosticModuleState.Ok, null, "test", DateTimeOffset.UtcNow, new Dictionary<string, string> { ["state"] = "Ready" }) }); };
        using var page = new DiagnosticPage();
        using var form = new Form { ClientSize = new Size(1000, 700) };
        var parent = new Panel { Dock = DockStyle.Fill };
        form.Controls.Add(parent);
        parent.Controls.Add(page);
        form.Show();
        page.OnShow();
        Pump(100);
        const BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic;
        var timer = (System.Windows.Forms.Timer)typeof(DiagnosticPage).GetField("_refreshTimer", flags)!.GetValue(page)!;
        var list = (ListView)typeof(DiagnosticPage).GetField("_modules", flags)!.GetValue(page)!;
        var details = (TextBox)typeof(DiagnosticPage).GetField("_details", flags)!.GetValue(page)!;
        list.Items[0].Selected = true;
        Pump(30);
        int beforeDetails = details.TextLength;
        typeof(DiagnosticPage).GetMethod("RefreshView", flags)!.Invoke(page, null);
        int afterDetails = details.TextLength;
        int beforeDetach = reads;
        parent.Controls.Clear();
        Pump(1200);
        Console.WriteLine(JsonSerializer.Serialize(new { beforeDetails, afterDetails, page.Visible, parentIsNull = page.Parent is null, timerEnabledAfterDetach = timer.Enabled, readsBeforeDetach = beforeDetach, readsAfterDetach = reads }));
        form.Close();
    }

    static void ApplyFault()
    {
        // Controlled fault after ThemeManager.Load, before geometry commit. No hardware is constructed.
        string name = "closure-same-name-" + Guid.NewGuid().ToString("N");
        string path = Path.Combine(ThemeManager.ThemeDir, name + ".json");
        Directory.CreateDirectory(ThemeManager.ThemeDir);
        File.WriteAllText(path, "{\"name\":\"Same Display Name\",\"version\":1}");
        var oldLoad = ThemeManager.LoadWithResult(name);
        Theme old = ThemeManager.Current;
        File.WriteAllText(path, "{\"name\":\"Same Display Name\",\"version\":2}");
        var ui = (UIController)System.Runtime.CompilerServices.RuntimeHelpers.GetUninitializedObject(typeof(UIController));
        var result = AppActions.ApplyThemeAndLayoutWithResult(new Settings { Skin = name }, ui, null!, wasHorizontal: false);
        Console.WriteLine(JsonSerializer.Serialize(new { result, currentChanged = !ReferenceEquals(old, ThemeManager.Current), sourceChanged = oldLoad.SourceHash != ThemeManager.LastLoad?.SourceHash, loadSucceeded = ThemeManager.LastLoad?.Success, currentVersion = ThemeManager.Current.Version }, new JsonSerializerOptions { WriteIndented = true }));
        old.DisposeFonts();
        ThemeManager.Current.DisposeFonts();
        File.Delete(path);
    }
