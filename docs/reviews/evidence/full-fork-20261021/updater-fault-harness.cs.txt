global using File = ReviewFaultFile;
global using Process = ReviewProcess;
global using Thread = ReviewThread;
using System.Reflection;
using System.IO.Compression;
using System.Text.Json;

internal static class UpdaterReviewHarness
{
    public static int Main()
    {
        string output = AppContext.BaseDirectory;
        string victim = Path.Combine(output, "review-payload.dat");
        System.IO.File.WriteAllText(Path.Combine(output, "LiteMonitor++.exe"), "review-placeholder");
        System.IO.File.WriteAllText(victim, "ORIGINAL-CONTENT");
        string zipPath = Path.Combine(output, "review-update.zip");
        using (var zip = ZipFile.Open(zipPath, ZipArchiveMode.Create))
        {
            using var writer = new StreamWriter(zip.CreateEntry("review-payload.dat").Open());
            writer.Write("UPDATED-CONTENT");
        }
        ReviewFaultFile.DestinationToFail = victim;
        var main = typeof(LiteMonitor.Updater.Program).GetMethod("Main", BindingFlags.Static | BindingFlags.NonPublic)!;
        int exitCode = (int)main.Invoke(null, new object[] { new[] { zipPath } })!;
        string current = System.IO.File.ReadAllText(victim);
        string backup = Directory.GetFiles(output, "review-payload.dat", SearchOption.AllDirectories)
            .First(p => p.Contains("_update_backup_"));
        var result = new
        {
            scenario = "partial-copy-failure-rollback",
            injection = "File.Copy writes partial destination then throws IOException; process launches and sleeps replaced with inert test shims",
            productionSource = "LiteMonitor.Updater/Program.cs (linked unchanged)",
            updaterExitCode = exitCode,
            copyFailures = ReviewFaultFile.CopyFailures,
            destinationContent = current,
            backupContent = System.IO.File.ReadAllText(backup),
            originalRestored = current == "ORIGINAL-CONTENT"
        };
        string json = JsonSerializer.Serialize(result, new JsonSerializerOptions { WriteIndented = true });
        System.IO.File.WriteAllText(Path.Combine(output, "results.json"), json);
        Console.WriteLine(json);
        return 0;
    }
}

internal static class ReviewFaultFile
{
    public static string? DestinationToFail;
    public static int CopyFailures;
    public static bool Exists(string? path) => System.IO.File.Exists(path);
    public static void Delete(string path) => System.IO.File.Delete(path);
    public static void WriteAllText(string path, string? contents) => System.IO.File.WriteAllText(path, contents);
    public static void AppendAllText(string path, string? contents) => System.IO.File.AppendAllText(path, contents);
    public static FileStream OpenRead(string path) => System.IO.File.OpenRead(path);
    public static void Copy(string source, string destination, bool overwrite)
    {
        if (string.Equals(destination, DestinationToFail, StringComparison.OrdinalIgnoreCase))
        {
            System.IO.File.WriteAllText(destination, "PARTIAL");
            CopyFailures++;
            throw new IOException("review: simulated partial write followed by IO failure");
        }
        System.IO.File.Copy(source, destination, overwrite);
    }
}
