using System.Drawing;
using System.Drawing.Imaging;
using System.Text.Json;
using LiteMonitor;
using LiteMonitor.src.Core;
using LiteMonitor.src.SystemServices.Backdrop;
using LiteMonitor.src.UI.Decor;
using LiteMonitor.src.UI.Helpers;
using LiteMonitor.ThemeEditor;
using LiteMonitor.AppearanceWiringProbe;

internal static class Program
{
    static string output = "";
    [STAThread]
    static void Main(string[] args)
    {
        output = Path.GetFullPath(args[0]);
        Directory.CreateDirectory(output);
        WiringFixture.InstallSettingsSingleton();
        var results = new Dictionary<string, object>();
        Case("preview", Preview);
        Case("noise_eviction", Noise);
        Case("bar_clip", Clip);
        Case("partial_background", PartialBackground);
        Case("off_allocations", Allocations);
        Case("texture_geometry", Texture);
        Case("resize_shape", Shape);
        Case("texture_image_ownership", ImageOwnership);
        Case("noise_pipeline", NoisePipeline);
        string json = JsonSerializer.Serialize(results, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(Path.Combine(output, args.Length > 1 ? args[1] + ".json" : "repros.json"), json);
        Console.WriteLine(json);
        void Case(string name, Func<object> check)
        {
            if (args.Length > 1 && name != args[1]) return;
            try { results[name] = check(); }
            catch (Exception ex) { results[name] = new { unexpectedException = ex.ToString() }; }
        }
    }

    static object Preview()
    {
        Theme theme = WiringFixture.LoadTheme("DarkNeo_Modern");
        theme.Layout.FrameStyle = 1;
        theme.Layout.FrameWidth = 1;
        theme.Layout.Glow = 0;
        theme.Layout.BarStyle = 0;
        theme.Color.Shadow = "#00FFFF";
        theme.Color.Border = "#00FFFF";
        var scene = WiringFixture.BuildPortrait(theme, 1);
        var bounds = new Rectangle(0, 0, scene.Width, scene.Height);
        using var runtime = new AppearanceRuntime();
        var result = new List<object>();
        foreach (string change in new[] { "Glow", "BarStyle" })
        {
            theme.Layout.Glow = 0; theme.Layout.BarStyle = 0;
            using Bitmap beforePreview = RenderPreview();
            runtime.OnThemeChanged(theme, 1, 1, bounds);
            using Bitmap beforeMain = WiringFixture.Render(scene, runtime);
            if (change == "Glow") theme.Layout.Glow = 100;
            else theme.Layout.BarStyle = 1;
            using Bitmap afterPreview = RenderPreview();
            runtime.OnThemeChanged(theme, 1, 1, bounds);
            using Bitmap afterMain = WiringFixture.Render(scene, runtime);
            afterPreview.Save(Path.Combine(output, "preview-" + change + ".png"));
            afterMain.Save(Path.Combine(output, "main-" + change + ".png"));
            result.Add(new {
                change,
                previewChangedPixels = WiringFixture.CountDifferentPixels(beforePreview, afterPreview),
                mainChangedPixels = WiringFixture.CountDifferentPixels(beforeMain, afterMain),
                beforePreviewHash = WiringFixture.HashPixels(beforePreview),
                afterPreviewHash = WiringFixture.HashPixels(afterPreview)
            });
        }
        theme.DisposeFonts();
        return result;
        Bitmap RenderPreview()
        {
            var bitmap = new Bitmap(scene.Width, scene.Height, PixelFormat.Format32bppPArgb);
            using Graphics g = Graphics.FromImage(bitmap);
            ThemePreviewControl.RenderPreview(g, scene.Groups, theme, bounds);
            return bitmap;
        }
    }

    static object Noise()
    {
        using var cache = new BackdropNoiseCache();
        using var bitmap = new Bitmap(32, 32);
        using Graphics g = Graphics.FromImage(bitmap);
        var steps = new List<object>();
        foreach (int intensity in new[] { 10, 10, 20, 20, 30, 30, 40, 40, 50 })
        {
            try
            {
                var brush = cache.GetBrush(intensity);
                g.FillRectangle(brush!, 0, 0, 32, 32);
                steps.Add(new { intensity, drawn = true, entryCount = cache.EntryCount });
            }
            catch (Exception ex)
            {
                steps.Add(new { intensity, drawn = false, exception = ex.GetType().FullName, message = ex.Message, entryCount = cache.EntryCount });
            }
        }
        return steps;
    }

    static object Clip()
    {
        var results = new List<object>();
        Rectangle clip = new(37, 43, 20, 20);
        foreach (int style in new[] { 1, 2 })
        {
            Theme theme = new(); theme.Layout.BarStyle = style;
            var bounds = new Rectangle(0, 0, 240, 100);
            using var runtime = new AppearanceRuntime();
            var options = runtime.Update(theme, 1, 1, bounds);
            using var bitmap = new Bitmap(bounds.Width, bounds.Height);
            using (Graphics g = Graphics.FromImage(bitmap))
            {
                g.Clear(Color.Magenta); g.SetClip(clip);
                runtime.Bars.RenderBarSlot(g, options, new Rectangle(20, 40, 180, 8), .5, Color.Cyan);
            }
            int outside = 0;
            for (int y = 0; y < bitmap.Height; y++)
                for (int x = 0; x < bitmap.Width; x++)
                    if (!clip.Contains(x, y) && bitmap.GetPixel(x, y).ToArgb() != Color.Magenta.ToArgb()) outside++;
            bitmap.Save(Path.Combine(output, $"bar-clip-style{style}.png"));
            results.Add(new { style, changedPixelsOutsideCallerClip = outside });
        }
        return results;
    }

    static object PartialBackground()
    {
        Theme theme = WiringFixture.LoadTheme("DarkNeo_Modern");
        theme.Color.Background = "#123456";
        var bounds = new Rectangle(0, 0, 240, 160);
        var results = new List<object>();
        foreach (bool horizontal in new[] { false, true })
        {
            using var runtime = new AppearanceRuntime();
            runtime.Update(theme, 1, 1, bounds);
            using var bitmap = new Bitmap(bounds.Width, bounds.Height);
            using (Graphics g = Graphics.FromImage(bitmap))
            {
                g.Clear(Color.Magenta); g.SetClip(new Rectangle(37, 43, 20, 20));
                if (horizontal) HorizontalRenderer.Render(g, theme, new List<Column>(), 240, runtime);
                else UIRenderer.Render(g, new List<GroupLayoutInfo>(), theme, runtime);
            }
            results.Add(new { horizontal, sampleArgb = bitmap.GetPixel(40, 45).ToArgb().ToString("X8"), expectedArgb = "FF123456" });
        }
        theme.DisposeFonts();
        return results;
    }

    static object Allocations()
    {
        Theme theme = WiringFixture.LoadTheme("DarkNeo_Modern");
        var scene = WiringFixture.BuildPortrait(theme, 1);
        using var runtime = new AppearanceRuntime();
        runtime.Update(theme, 1, 1, new Rectangle(0, 0, scene.Width, scene.Height));
        using var bitmap = new Bitmap(scene.Width, scene.Height);
        using Graphics g = Graphics.FromImage(bitmap);
        for (int i = 0; i < 200; i++) { Draw(false); Draw(true); }
        var samples = new List<object>();
        for (int run = 0; run < 3; run++)
        {
            long oldBytes = Measure(false), newBytes = Measure(true);
            samples.Add(new { run, oldBytesPerFrame = oldBytes / 500.0, newBytesPerFrame = newBytes / 500.0, addedBytesPerFrame = (newBytes - oldBytes) / 500.0 });
        }
        theme.DisposeFonts();
        return samples;
        void Draw(bool enabled) => UIRenderer.Render(g, scene.Groups, theme, enabled ? runtime : null);
        long Measure(bool enabled)
        {
            long start = GC.GetAllocatedBytesForCurrentThread();
            for (int i = 0; i < 500; i++) Draw(enabled);
            return GC.GetAllocatedBytesForCurrentThread() - start;
        }
    }

    static object Texture()
    {
        using var tile = DecorTextureFactory.CreateBarTextureTile(16, 16, Color.Cyan)!;
        tile.Save(Path.Combine(output, "bar-texture-tile.png"));
        int interiorPixels = 0, totalPixels = 0;
        for (int y = 0; y < 16; y++)
            for (int x = 0; x < 16; x++)
                if (tile.GetPixel(x, y).A > 0) { totalPixels++; if (x > 1 && x < 14 && y > 1 && y < 14) interiorPixels++; }
        return new { size = 16, nonTransparentPixels = totalPixels, interiorNonTransparentPixels = interiorPixels, expectedInteriorPattern = true };
    }

    static object Shape()
    {
        Theme theme = new(); theme.Layout.FrameStyle = 1; theme.Layout.FrameLength = 64;
        using var form = new Form { FormBorderStyle = FormBorderStyle.None, ShowInTaskbar = false, StartPosition = FormStartPosition.Manual, Location = new Point(-32000, -32000), ClientSize = new Size(240, 80) };
        _ = form.Handle;
        using var shape = new WindowAppearanceController(form);
        var oldOptions = ThemeAppearanceOptions.Create(theme, 1, 1, form.ClientRectangle, 0);
        shape.Apply(oldOptions, 8);
        form.ClientSize = new Size(240, 200);
        shape.Reapply();
        var newOptions = ThemeAppearanceOptions.Create(theme, 1, 1, form.ClientRectangle, 0);
        bool oldRegionIncludesPoint = form.Region!.IsVisible(50, 5);
        int staleLength = shape.LastCornerLengthPx;
        shape.Apply(newOptions, 8);
        return new { oldHeight = 80, newHeight = 200, oldSnapshotLength = oldOptions.FrameLengthPx, lengthAfterReapply = staleLength, freshSnapshotLength = newOptions.FrameLengthPx, pointX = 50, pointY = 5, staleRegionIncludesPoint = oldRegionIncludesPoint, freshRegionIncludesPoint = form.Region!.IsVisible(50, 5) };
    }

    static object ImageOwnership()
    {
        using var source = new Bitmap(16, 16);
        using var brush = new TextureBrush(source);
        using Image first = brush.Image;
        using Image second = brush.Image;
        bool retrievedIsSource = ReferenceEquals(first, source);
        bool successiveImagesAreSame = ReferenceEquals(first, second);
        source.Dispose();
        first.Dispose();
        int secondWidthAfterOtherDisposals = second.Width;
        using var canvas = new Bitmap(16, 16);
        using Graphics g = Graphics.FromImage(canvas);
        g.FillRectangle(brush, 0, 0, 16, 16);
        return new { retrievedIsSource, successiveImagesAreSame, secondWidthAfterOtherDisposals, brushDrawsAfterSourceAndRetrievedImageDisposed = true };
    }

    static object NoisePipeline()
    {
        Theme theme = WiringFixture.LoadTheme("DarkNeo_Modern");
        theme.Backdrop = new BackdropConfig { Mode = "gradient" };
        using var runtime = new AppearanceRuntime();
        using var bitmap = new Bitmap(240, 160);
        using Graphics g = Graphics.FromImage(bitmap);
        var steps = new List<object>();
        foreach (int intensity in new[] { 10, 10, 20, 20, 30, 30, 40, 40, 50 })
        {
            theme.Backdrop.Noise = intensity;
            runtime.OnThemeChanged(theme, 1, 1, new Rectangle(0, 0, 240, 160));
            try
            {
                UIRenderer.Render(g, new List<GroupLayoutInfo>(), theme, runtime);
                steps.Add(new { intensity, drawn = true });
            }
            catch (Exception ex)
            {
                steps.Add(new { intensity, drawn = false, exception = ex.ToString() });
            }
        }
        theme.DisposeFonts();
        return steps;
    }
}
