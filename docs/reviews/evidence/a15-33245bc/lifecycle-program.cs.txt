using System.Diagnostics;
using System.Drawing;
using System.Text.Json;
using LiteMonitor.src.SystemServices.Backdrop;

internal static class Program
{
    [STAThread]
    static void Main(string[] args)
    {
        var results = new Dictionary<string, object>();
        Check("stop_timeout_restart", Restart);
        Check("invalid_active_borrow", InvalidActive);
        Check("suspend_inflight", Pause);
        Check("success_state_text", StateText);
        Directory.CreateDirectory(args[0]);
        string json = JsonSerializer.Serialize(results, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(Path.Combine(args[0], "lifecycle-repros.json"), json);
        Console.WriteLine(json);
        void Check(string name, Func<object> test)
        {
            try { results[name] = test(); }
            catch (Exception ex) { results[name] = new { unexpectedException = ex.ToString() }; }
        }
    }

    static BackdropServiceRequest Request(int generation = 1) => new(
        IntPtr.Zero, new Rectangle(0, 0, 16, 16), 1f, 1f,
        generation, generation, generation, 0, targetFps: 1, captureScale: 1);

    static bool Until(Func<bool> condition, int timeoutMs = 2500)
    {
        var watch = Stopwatch.StartNew();
        while (watch.ElapsedMilliseconds < timeoutMs)
        {
            if (condition()) return true;
            Thread.Sleep(2);
        }
        return condition();
    }

    static object Restart()
    {
        using var backend = new Backend { BlockFirst = true, BlockSecond = true };
        using var service = new BackdropCaptureService(backend, takeoverTimeoutMs: 10);
        BackdropFrameLease? lease = null;
        try
        {
            var request = Request();
            service.Start(request);
            if (!backend.FirstEntered.Wait(1500)) throw new Exception("First worker did not reach gate.");
            bool stopped = service.Stop(TimeSpan.FromMilliseconds(20));
            service.Start(request);
            bool secondEntered = backend.SecondEntered.Wait(1500);
            backend.FirstRelease.Set();
            bool latePending = Until(() => service.PendingFrameCount > 0);
            bool acquired = service.TryAcquireFrame(out lease);
            string? color = acquired ? ((Bitmap)lease!.Frame).GetPixel(0, 0).ToArgb().ToString("X8") : null;
            bool oldWorkerSetStopped = Until(() => service.Status.State == BackdropServiceState.Stopped, 500);
            return new { stopCompleted = stopped, secondEntered, maxConcurrentBackendCalls = backend.MaxConcurrent, latePending, acquired, lateFrameArgb = color, redMeansOldWorker = color == "FFFF0000", oldWorkerSetStopped, isNewWorkerStillRunning = service.IsRunning, state = service.Status.State.ToString() };
        }
        finally
        {
            backend.FirstRelease.Set(); backend.SecondRelease.Set();
            service.Stop(TimeSpan.FromSeconds(2));
            lease?.Dispose();
        }
    }

    static object InvalidActive()
    {
        using var backend = new Backend();
        using var service = new BackdropCaptureService(backend, takeoverTimeoutMs: 5);
        BackdropFrameLease? lease = null;
        try
        {
            service.Start(Request());
            if (!Until(() => service.PendingFrameCount > 0) || !service.TryAcquireFrame(out lease)) throw new Exception("No initial lease.");
            service.UpdateRequest(Request(2));
            var afterGeneration = Snapshot();
            service.SetSuspended(true);
            var afterSuspension = Snapshot();
            service.Stop(TimeSpan.FromSeconds(2));
            var afterStop = Snapshot();
            service.Dispose();
            var afterDispose = Snapshot();
            return new { afterGeneration, afterSuspension, afterStop, afterDispose };
        }
        finally { service.Stop(TimeSpan.FromSeconds(2)); lease?.Dispose(); }
        object Snapshot() => new { leaseValid = lease!.IsValid, borrowed = service.TryGetFrame(0, out _), ready = service.IsReady, state = service.Status.State.ToString(), effectiveMode = service.Status.EffectiveMode.ToString() };
    }

    static object Pause()
    {
        using var backend = new Backend { BlockFirst = true };
        using var service = new BackdropCaptureService(backend, takeoverTimeoutMs: 5);
        BackdropFrameLease? lease = null;
        try
        {
            service.Start(Request());
            if (!backend.FirstEntered.Wait(1500)) throw new Exception("Capture did not enter gate.");
            service.SetSuspended(true);
            string stateAtPause = service.Status.State.ToString();
            backend.FirstRelease.Set();
            bool pendingWhileSuspended = Until(() => service.PendingFrameCount > 0);
            bool acquiredWhileSuspended = service.TryAcquireFrame(out lease);
            Until(() => service.Metrics.FramesPublished > 0, 500);
            return new { stateAtPause, pendingWhileSuspended, acquiredWhileSuspended, framesPublished = service.Metrics.FramesPublished, stateAfterCompletion = service.Status.State.ToString() };
        }
        finally { backend.FirstRelease.Set(); service.Stop(TimeSpan.FromSeconds(2)); lease?.Dispose(); }
    }

    static object StateText()
    {
        using var backend = new Backend { FailFirst = true };
        using var service = new BackdropCaptureService(backend, takeoverTimeoutMs: 5);
        try
        {
            service.Start(Request());
            if (!Until(() => service.Metrics.CaptureFailures > 0)) throw new Exception("Expected failure not observed.");
            string? failureText = service.Status.StateText;
            bool recovered = Until(() => service.Metrics.FramesPublished > 0, 5000);
            var status = service.Status;
            return new { recovered, state = status.State.ToString(), reason = status.DegradeReason.ToString(), failureText, currentText = status.StateText, stillShowsFailure = status.StateText == failureText };
        }
        finally { service.Stop(TimeSpan.FromSeconds(2)); }
    }

    sealed class Backend : IBackdropCaptureBackend, IDisposable
    {
        public bool BlockFirst, BlockSecond, FailFirst;
        public readonly ManualResetEventSlim FirstEntered = new(), SecondEntered = new();
        public readonly ManualResetEventSlim FirstRelease = new(), SecondRelease = new();
        int calls, concurrent, maximum;
        public int MaxConcurrent => Volatile.Read(ref maximum);
        public bool IsReady => true;
        public string UnavailableReason => "synthetic first-capture failure";
        public bool IsPreview => false;
        public bool TryCapture(BackdropServiceRequest request, out Image? frame, out BackdropServiceTiming timing)
        {
            int call = Interlocked.Increment(ref calls);
            int active = Interlocked.Increment(ref concurrent);
            int previous;
            do { previous = Volatile.Read(ref maximum); if (active <= previous) break; }
            while (Interlocked.CompareExchange(ref maximum, active, previous) != previous);
            try
            {
                if (call == 1) { FirstEntered.Set(); if (BlockFirst) FirstRelease.Wait(8000); }
                if (call == 2) { SecondEntered.Set(); if (BlockSecond) SecondRelease.Wait(8000); }
                timing = new BackdropServiceTiming(excludeWindowMs: 0, backendCaptureMs: 1, backendFilterMs: 0);
                if (call == 1 && FailFirst) { frame = null; return false; }
                var bitmap = new Bitmap(16, 16);
                bitmap.SetPixel(0, 0, call == 1 ? Color.Red : Color.Blue);
                frame = bitmap;
                return true;
            }
            finally { Interlocked.Decrement(ref concurrent); }
        }
        public void RestoreDisplayAffinity() { }
        public void Dispose()
        {
            FirstRelease.Set(); SecondRelease.Set();
            FirstEntered.Dispose(); SecondEntered.Dispose(); FirstRelease.Dispose(); SecondRelease.Dispose();
        }
    }
}
